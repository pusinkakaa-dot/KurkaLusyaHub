local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer

-- toggle состояние
if player:GetAttribute("NameGuiToggle") == nil then
	player:SetAttribute("NameGuiToggle", false)
end

local textLabels = {}

-- создаём GUI
local function addGui(plr, char)
	if plr == player then return end
	if not player:GetAttribute("NameGuiToggle") then return end

	local head = char:FindFirstChild("Head")
	if not head then return end

	if head:FindFirstChild("NameGui") then return end

	local billboard = Instance.new("BillboardGui")
	billboard.Name = "NameGui"
	billboard.Size = UDim2.new(0, 200, 0, 50)
	billboard.StudsOffset = Vector3.new(0, 1, 0)
	billboard.AlwaysOnTop = true
	billboard.Parent = head

	local text = Instance.new("TextLabel")
	text.Size = UDim2.new(1, 0, 1, 0)
	text.BackgroundTransparency = 1
	text.Text = plr.Name

	text.TextScaled = false
	text.TextSize = 14
	text.TextColor3 = Color3.new(1, 1, 1)
	text.Font = Enum.Font.SourceSansBold
	text.Parent = billboard

	table.insert(textLabels, text)
end

-- удаляем GUI
local function removeGui(char)
	local head = char:FindFirstChild("Head")
	if not head then return end

	local gui = head:FindFirstChild("NameGui")
	if gui then
		gui:Destroy()
	end
end

-- применяем ко всем
local function apply(state)
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= player and plr.Character then
			if state then
				addGui(plr, plr.Character)
			else
				removeGui(plr, plr.Character)
			end
		end
	end
end

-- 🔁 toggle
local toggled = player:GetAttribute("NameGuiToggle")
local newState = not toggled
player:SetAttribute("NameGuiToggle", newState)
print(newState and "Text (nicknames) above the heads of all players except you: 🟢 Enabled" or "Text (nicknames) above the heads of all players except you: 🔴 Disabled")

-- полная очистка при выключении
if not newState then
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= player and plr.Character then
			removeGui(plr.Character)
		end
	end
	table.clear(textLabels)
end

apply(newState)

-- 🌈 радуга
local hue = 0

RunService.RenderStepped:Connect(function(dt)
	hue += dt * 0.2
	if hue > 1 then hue = 0 end

	local color = Color3.fromHSV(hue, 1, 1)

	for i = #textLabels, 1, -1 do
		local text = textLabels[i]
		if text and text.Parent then
			text.TextColor3 = color
		else
			table.remove(textLabels, i)
		end
	end
end)

-- 🔥 проверка каждые 3 секунды
task.spawn(function()
	while true do
		if player:GetAttribute("NameGuiToggle") then
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= player and plr.Character then
					addGui(plr, plr.Character)
				end
			end
		end
		task.wait(3)
	end
end)
