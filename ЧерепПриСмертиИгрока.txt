local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local TweenService = game:GetService("TweenService")

local LocalPlayer = Players.LocalPlayer
if not LocalPlayer then return end

-- Настройки смайлика черепа после смерти
local SKULL_EMOJI = "💀"
local DISPLAY_DURATION = 2  -- Время показа в секундах
local SPIN_SPEED_DEG = 720    -- Скорость вращения в градусах за секунду (2 оборота в сек)

-- Размер вблизи совпадает с размером кольца (1.15 студа) и растёт с дистанцией
local RING_MATCH_STUDS = 1.15 -- Базовый размер вблизи (как кольцо)
local MAX_SKULL_STUDS = 14.0  -- Большой размер при сильном удалении
local MIN_DISTANCE = 12       -- Дистанция (в студах), до которой размер равен RING_MATCH_STUDS
local MAX_DISTANCE = 220      -- Дистанция (в студах), на которой размер достигает MAX_SKULL_STUDS

-- Переключатель Вкл / Выкл при каждом запуске
local TOGGLE_ATTR = "DeathSkullSpinState"
local RUN_ID_ATTR = "DeathSkullSpinRunId"

local stateFolder = ReplicatedStorage:FindFirstChild("DeathSkullSpinFolder")
if not stateFolder then
	stateFolder = Instance.new("Folder")
	stateFolder.Name = "DeathSkullSpinFolder"
	stateFolder.Parent = ReplicatedStorage
end

local nextState = not (stateFolder:GetAttribute(TOGGLE_ATTR) == true)
local runId = (stateFolder:GetAttribute(RUN_ID_ATTR) or 0) + 1

stateFolder:SetAttribute(TOGGLE_ATTR, nextState)
stateFolder:SetAttribute(RUN_ID_ATTR, runId)

print("Appearance of a skull on players who die (2-second delay): " .. (nextState and "🟢 Enabled" or "🔴 Disabled"))

local function isActiveRun(): boolean
	return stateFolder:GetAttribute(TOGGLE_ATTR) == true
		and stateFolder:GetAttribute(RUN_ID_ATTR) == runId
end

local connections: { RBXScriptConnection } = {}
local activeAnchors: { BasePart } = {}
local activeLabels: { TextLabel } = {}

local function clearAllMarkers()
	table.clear(activeLabels)

	for _, anchor in ipairs(activeAnchors) do
		if anchor and anchor.Parent then
			anchor:Destroy()
		end
	end
	table.clear(activeAnchors)

	for _, child in ipairs(workspace:GetChildren()) do
		if child.Name == "DeathSkullSpinAnchorPart" then
			child:Destroy()
		end
	end
end

local function stopAll()
	for _, conn in ipairs(connections) do
		conn:Disconnect()
	end
	table.clear(connections)
	clearAllMarkers()
end

if not nextState then
	stopAll()
	return
end

clearAllMarkers()

-- Плавное и быстрое вращение всех активных черепов
table.insert(connections, RunService.RenderStepped:Connect(function(dt: number)
	if not isActiveRun() then
		stopAll()
		return
	end

	for i = #activeLabels, 1, -1 do
		local label = activeLabels[i]
		if label and label.Parent then
			label.Rotation = (label.Rotation + SPIN_SPEED_DEG * dt) % 360
		else
			table.remove(activeLabels, i)
		end
	end
end))

-- Вычисляем размер черепа в студах: вблизи = как кольцо, вдали = большой
local function getSkullSizeStuds(deathPosition: Vector3): number
	local myChar = LocalPlayer.Character
	local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart") :: BasePart?
	if not myRoot then
		return RING_MATCH_STUDS
	end

	local distance = (myRoot.Position - deathPosition).Magnitude
	local alpha = math.clamp((distance - MIN_DISTANCE) / (MAX_DISTANCE - MIN_DISTANCE), 0, 1)
	return RING_MATCH_STUDS + (MAX_SKULL_STUDS - RING_MATCH_STUDS) * alpha
end

-- Получаем позицию туловища персонажа (R15 / R6)
local function getTorsoPosition(character: Model): Vector3
	local torsoParts: { BasePart } = {}
	for _, partName in ipairs({ "UpperTorso", "LowerTorso", "Torso" }) do
		local part = character:FindFirstChild(partName)
		if part and part:IsA("BasePart") then
			table.insert(torsoParts, part)
		end
	end

	if #torsoParts > 0 then
		local sum = Vector3.zero
		for _, part in ipairs(torsoParts) do
			sum += part.Position
		end
		return sum / #torsoParts
	end

	local fallback = (character:FindFirstChild("HumanoidRootPart") or character.PrimaryPart) :: BasePart?
	return fallback and fallback.Position or character:GetPivot().Position
end

-- Резко показываем вращающийся череп на туловище в момент смерти
local function spawnDeathSkull(torsoPosition: Vector3)
	if not isActiveRun() then return end

	local sizeStuds = getSkullSizeStuds(torsoPosition)

	local anchor = Instance.new("Part")
	anchor.Name = "DeathSkullSpinAnchorPart"
	anchor.Size = Vector3.new(0.2, 0.2, 0.2)
	anchor.Position = torsoPosition
	anchor.Anchored = true
	anchor.CanCollide = false
	anchor.CanQuery = false
	anchor.CanTouch = false
	anchor.Transparency = 1
	anchor.Parent = workspace

	table.insert(activeAnchors, anchor)

	local billboard = Instance.new("BillboardGui")
	billboard.Name = "DeathSkullSpinGUI"
	billboard.Adornee = anchor
	billboard.AlwaysOnTop = true
	billboard.Size = UDim2.fromScale(sizeStuds, sizeStuds)
	billboard.MaxDistance = 2500
	billboard.Parent = anchor

	local label = Instance.new("TextLabel")
	label.Name = "SkullEmojiLabel"
	label.AnchorPoint = Vector2.new(0.5, 0.5)
	label.Position = UDim2.fromScale(0.5, 0.5)
	label.Size = UDim2.fromScale(0.25, 0.25)
	label.BackgroundTransparency = 1
	label.Text = SKULL_EMOJI
	label.TextScaled = true
	label.Font = Enum.Font.GothamBlack
	label.Parent = billboard

	table.insert(activeLabels, label)

	-- Резкий рывок размера при появлении
	local popTween = TweenService:Create(
		label,
		TweenInfo.new(0.1, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
		{ Size = UDim2.fromScale(1, 1) }
	)
	popTween:Play()

	-- Удаляем по истечении времени показа
	task.delay(DISPLAY_DURATION, function()
		if anchor and anchor.Parent then
			anchor:Destroy()
		end
	end)
end

-- Отслеживаем полную смерть конкретного персонажа
local function trackCharacterDeath(player: Player, character: Model)
	if player == LocalPlayer then return end

	local hum = character:WaitForChild("Humanoid", 5) :: Humanoid?
	if not hum or not isActiveRun() then return end

	local triggered = false
	local function onDeath()
		if triggered or not isActiveRun() then return end
		triggered = true

		spawnDeathSkull(getTorsoPosition(character))
	end

	table.insert(connections, hum.Died:Connect(onDeath))
	table.insert(connections, hum.HealthChanged:Connect(function(hp)
		if hp <= 0 then
			onDeath()
		end
	end))
end

-- Подключаемся ко всем текущим и новым игрокам
local function setupPlayer(player: Player)
	if player == LocalPlayer then return end

	if player.Character then
		task.spawn(trackCharacterDeath, player, player.Character)
	end

	table.insert(connections, player.CharacterAdded:Connect(function(char)
		if isActiveRun() then
			trackCharacterDeath(player, char)
		end
	end))
end

for _, plr in ipairs(Players:GetPlayers()) do
	setupPlayer(plr)
end

table.insert(connections, Players.PlayerAdded:Connect(function(plr)
	if isActiveRun() then
		setupPlayer(plr)
	end
end))
