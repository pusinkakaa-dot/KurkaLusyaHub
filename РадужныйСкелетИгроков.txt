local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")

local LocalPlayer = Players.LocalPlayer
if not LocalPlayer then return end

-- Настройки радужного скелета
local LINE_THICKNESS = 2
local LINE_TRANSPARENCY = 0.9 -- В Drawing API: 1 = непрозрачная, 0 = невидимая
local RAINBOW_SPEED = 1.5
local MAX_DISTANCE = 2500

-- Переключатель Вкл / Выкл через ReplicatedStorage
local TOGGLE_ATTR = "RainbowSkeletonState"
local RUN_ID_ATTR = "RainbowSkeletonRunId"

local stateFolder = ReplicatedStorage:FindFirstChild("RainbowSkeletonFolder")
if not stateFolder then
	stateFolder = Instance.new("Folder")
	stateFolder.Name = "RainbowSkeletonFolder"
	stateFolder.Parent = ReplicatedStorage
end

local nextState = not (stateFolder:GetAttribute(TOGGLE_ATTR) == true)
local runId = (stateFolder:GetAttribute(RUN_ID_ATTR) or 0) + 1

stateFolder:SetAttribute(TOGGLE_ATTR, nextState)
stateFolder:SetAttribute(RUN_ID_ATTR, runId)

print(nextState and "The skeletons of all players except you are shown: 🟢 Enabled " or "The skeletons of all players except you are shown: 🔴 Disabled")

local function isActiveRun(): boolean
	return stateFolder:GetAttribute(TOGGLE_ATTR) == true
		and stateFolder:GetAttribute(RUN_ID_ATTR) == runId
end

-- Соединения костей для R15
local R15_BONES = {
	{ "Head", "UpperTorso" },
	{ "UpperTorso", "LowerTorso" },
	{ "UpperTorso", "LeftUpperArm" },
	{ "LeftUpperArm", "LeftLowerArm" },
	{ "LeftLowerArm", "LeftHand" },
	{ "UpperTorso", "RightUpperArm" },
	{ "RightUpperArm", "RightLowerArm" },
	{ "RightLowerArm", "RightHand" },
	{ "LowerTorso", "LeftUpperLeg" },
	{ "LeftUpperLeg", "LeftLowerLeg" },
	{ "LeftLowerLeg", "LeftFoot" },
	{ "LowerTorso", "RightUpperLeg" },
	{ "RightUpperLeg", "RightLowerLeg" },
	{ "RightLowerLeg", "RightFoot" },
}

-- Узлы и кости для аккуратного скелета R6
local R6_BONES = {
	{ "Head", "R6_Neck" },
	{ "R6_Neck", "R6_LeftShoulder" },
	{ "R6_Neck", "R6_RightShoulder" },
	{ "R6_Neck", "R6_Pelvis" },
	{ "R6_LeftShoulder", "R6_LeftHand" },
	{ "R6_RightShoulder", "R6_RightHand" },
	{ "R6_Pelvis", "R6_LeftHip" },
	{ "R6_Pelvis", "R6_RightHip" },
	{ "R6_LeftHip", "R6_LeftFoot" },
	{ "R6_RightHip", "R6_RightFoot" },
}

local playerLines: { [Player]: { { line: Drawing, a: string, b: string } } } = {}
local connections: { RBXScriptConnection } = {}

local function getRainbowColor(offset: number): Color3
	local t = tick() * RAINBOW_SPEED + offset
	local r = math.sin(t + 0) * 127 + 128
	local g = math.sin(t + 2) * 127 + 128
	local b = math.sin(t + 4) * 127 + 128
	return Color3.fromRGB(math.clamp(math.floor(r), 0, 255), math.clamp(math.floor(g), 0, 255), math.clamp(math.floor(b), 0, 255))
end

local function clearPlayerLines(plr: Player)
	local entries = playerLines[plr]
	if entries then
		for _, item in ipairs(entries) do
			pcall(function()
				item.line.Visible = false
				item.line:Remove()
			end)
		end
		playerLines[plr] = nil
	end
end

local function stopAll()
	for _, conn in ipairs(connections) do
		conn:Disconnect()
	end
	table.clear(connections)

	for plr in pairs(playerLines) do
		clearPlayerLines(plr)
	end
end

if not nextState then
	stopAll()
	return
end

stopAll()

local function getR6Points(char: Model): { [string]: Vector3 }?
	local torso = char:FindFirstChild("Torso") :: BasePart?
	local head = char:FindFirstChild("Head") :: BasePart?
	if not torso or not head then return nil end

	local leftArm = char:FindFirstChild("Left Arm") :: BasePart?
	local rightArm = char:FindFirstChild("Right Arm") :: BasePart?
	local leftLeg = char:FindFirstChild("Left Leg") :: BasePart?
	local rightLeg = char:FindFirstChild("Right Leg") :: BasePart?

	local tCF = torso.CFrame
	local points: { [string]: Vector3 } = {
		Head = head.Position,
		R6_Neck = (tCF * CFrame.new(0, torso.Size.Y * 0.45, 0)).Position,
		R6_LeftShoulder = (tCF * CFrame.new(-torso.Size.X * 0.5, torso.Size.Y * 0.35, 0)).Position,
		R6_RightShoulder = (tCF * CFrame.new(torso.Size.X * 0.5, torso.Size.Y * 0.35, 0)).Position,
		R6_Pelvis = (tCF * CFrame.new(0, -torso.Size.Y * 0.45, 0)).Position,
		R6_LeftHip = (tCF * CFrame.new(-torso.Size.X * 0.25, -torso.Size.Y * 0.45, 0)).Position,
		R6_RightHip = (tCF * CFrame.new(torso.Size.X * 0.25, -torso.Size.Y * 0.45, 0)).Position,
	}

	if leftArm then
		points.R6_LeftHand = (leftArm.CFrame * CFrame.new(0, -leftArm.Size.Y * 0.45, 0)).Position
	end
	if rightArm then
		points.R6_RightHand = (rightArm.CFrame * CFrame.new(0, -rightArm.Size.Y * 0.45, 0)).Position
	end
	if leftLeg then
		points.R6_LeftFoot = (leftLeg.CFrame * CFrame.new(0, -leftLeg.Size.Y * 0.45, 0)).Position
	end
	if rightLeg then
		points.R6_RightFoot = (rightLeg.CFrame * CFrame.new(0, -rightLeg.Size.Y * 0.45, 0)).Position
	end

	return points
end

local function getBonePoint(char: Model, name: string, r6Points: { [string]: Vector3 }?): Vector3?
	if r6Points and r6Points[name] then
		return r6Points[name]
	end
	local part = char:FindFirstChild(name) :: BasePart?
	return part and part.Position or nil
end

local function ensurePlayerLines(plr: Player, char: Model)
	if playerLines[plr] then
		return playerLines[plr]
	end

	local isR15 = char:FindFirstChild("UpperTorso") ~= nil
	local bones = isR15 and R15_BONES or R6_BONES
	local list = {}

	for _, pair in ipairs(bones) do
		local line = Drawing.new("Line")
		line.Thickness = LINE_THICKNESS
		line.Transparency = LINE_TRANSPARENCY
		line.Visible = false
		table.insert(list, {
			line = line,
			a = pair[1],
			b = pair[2],
		})
	end

	playerLines[plr] = list
	return list
end

local function hidePlayerLines(plr: Player)
	local entries = playerLines[plr]
	if not entries then return end
	for _, item in ipairs(entries) do
		item.line.Visible = false
	end
end

table.insert(connections, Players.PlayerRemoving:Connect(function(plr)
	clearPlayerLines(plr)
end))

table.insert(connections, Players.PlayerAdded:Connect(function(plr)
	if plr == LocalPlayer then return end
	table.insert(connections, plr.CharacterAdded:Connect(function()
		clearPlayerLines(plr)
	end))
end))

for _, plr in ipairs(Players:GetPlayers()) do
	if plr ~= LocalPlayer then
		table.insert(connections, plr.CharacterAdded:Connect(function()
			clearPlayerLines(plr)
		end))
	end
end

table.insert(connections, RunService.RenderStepped:Connect(function()
	if not isActiveRun() then
		stopAll()
		return
	end

	local camera = workspace.CurrentCamera
	if not camera then return end

	local camPos = camera.CFrame.Position

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr == LocalPlayer then continue end

		local char = plr.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local root = char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")) :: BasePart?

		if not char or not hum or hum.Health <= 0 or not root then
			hidePlayerLines(plr)
			continue
		end

		if (root.Position - camPos).Magnitude > MAX_DISTANCE then
			hidePlayerLines(plr)
			continue
		end

		local isR15 = char:FindFirstChild("UpperTorso") ~= nil
		local r6Points = not isR15 and getR6Points(char) or nil
		local bonesList = ensurePlayerLines(plr, char)

		for idx, item in ipairs(bonesList) do
			local posA3D = getBonePoint(char, item.a, r6Points)
			local posB3D = getBonePoint(char, item.b, r6Points)

			if posA3D and posB3D then
				local posA, onScreenA = camera:WorldToViewportPoint(posA3D)
				local posB, onScreenB = camera:WorldToViewportPoint(posB3D)

				if onScreenA and onScreenB and posA.Z > 0 and posB.Z > 0 then
					item.line.From = Vector2.new(posA.X, posA.Y)
					item.line.To = Vector2.new(posB.X, posB.Y)
					item.line.Color = getRainbowColor(idx * 0.15)
					item.line.Visible = true
				else
					item.line.Visible = false
				end
			else
				item.line.Visible = false
			end
		end
	end
end))
