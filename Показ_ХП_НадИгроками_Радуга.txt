local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer

-- toggle состояние
if player:GetAttribute("HpGuiToggle") == nil then
	player:SetAttribute("HpGuiToggle", false)
end

local connections = {}
local textLabels = {}

-- создаём GUI с HP
local function addGui(plr, char)
	if plr == player then return end
	if not player:GetAttribute("HpGuiToggle") then return end

	local humanoid = char:FindFirstChildOfClass("Humanoid")
	local root = char:FindFirstChild("HumanoidRootPart")
	if not humanoid or not root then return end

	if root:FindFirstChild("HpGui") then return end

	local billboard = Instance.new("BillboardGui")
	billboard.Name = "HpGui"
	billboard.Size = UDim2.new(0, 200, 0, 40)
	billboard.StudsOffset = Vector3.new(0, -3, 0)
	billboard.AlwaysOnTop = true
	billboard.Parent = root

	local text = Instance.new("TextLabel")
	text.Size = UDim2.new(1, 0, 1, 0)
	text.BackgroundTransparency = 1
	text.TextScaled = false
	text.TextSize = 14
	text.Font = Enum.Font.SourceSansBold
	text.TextColor3 = Color3.new(1, 1, 1)
	text.Parent = billboard

	-- обновление HP
	local function update()
		text.Text = math.floor(humanoid.Health) .. " | HP"
	end

	update()

	local conn = humanoid.HealthChanged:Connect(update)
	table.insert(connections, conn)
	table.insert(textLabels, text)
end

-- удаляем GUI
local function removeGui(char)
	local root = char:FindFirstChild("HumanoidRootPart")
	if not root then return end

	local gui = root:FindFirstChild("HpGui")
	if gui then
		gui:Destroy()
	end
end

-- применяем ко всем
local function apply(state)
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= player and plr.Character then
			if state then
				addGui(plr, plr.Character)
			else
				removeGui(plr.Character)
			end
		end
	end
end

-- 🔁 toggle
local toggled = player:GetAttribute("HpGuiToggle")
local newState = not toggled
player:SetAttribute("HpGuiToggle", newState)
print(newState and "Health text appears beneath the feet of all players except you: 🟢 Enabled" or "Health text appears beneath the feet of all players except you: 🔴 Disabled")

-- 🔥 полная очистка при выключении
if not newState then
	for _, conn in ipairs(connections) do
		conn:Disconnect()
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= player and plr.Character then
			removeGui(plr.Character)
		end
	end

	table.clear(connections)
	table.clear(textLabels)
end

apply(newState)

-- 🌈 радуга
local hue = 0

RunService.RenderStepped:Connect(function(dt)
	hue += dt * 0.2
	if hue > 1 then hue = 0 end

	local color = Color3.fromHSV(hue, 1, 1)

	for i = #textLabels, 1, -1 do
		local text = textLabels[i]
		if text and text.Parent then
			text.TextColor3 = color
		else
			table.remove(textLabels, i)
		end
	end
end)

-- 🔥 авто-проверка каждые 3 секунды
task.spawn(function()
	while true do
		if player:GetAttribute("HpGuiToggle") then
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= player and plr.Character then
					addGui(plr, plr.Character)
				end
			end
		end
		task.wait(3)
	end
end)
