local FillTransparency = 0.5
local OutlineColor = Color3.fromRGB(255, 255, 255)
local OutlineTransparency = 0
local DepthMode = Enum.HighlightDepthMode.AlwaysOnTop

local CoreGui = game:GetService("CoreGui")
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local lp = Players.LocalPlayer

if not lp then return end

-- Глобальные переменные
local playerHighlights = {}
local playerConnections = {}
local globalConnections = {}
local rainbowConnection

-- Функция полной очистки
local function cleanupAll()
	if rainbowConnection then
		rainbowConnection:Disconnect()
		rainbowConnection = nil
	end

	for _, conn in ipairs(globalConnections) do
		if conn then
			conn:Disconnect()
		end
	end
	table.clear(globalConnections)

	for plr, highlight in pairs(playerHighlights) do
		if highlight then
			highlight:Destroy()
		end
		playerHighlights[plr] = nil
	end

	for plr, conn in pairs(playerConnections) do
		if conn then
			conn:Disconnect()
		end
		playerConnections[plr] = nil
	end

	local storage = CoreGui:FindFirstChild("HighlightToggle")
	if storage then
		storage:Destroy()
	end
end

-- Проверяем toggle состояние
local toggleFolder = CoreGui:FindFirstChild("HighlightToggle")
if toggleFolder then
	-- Состояние ВКЛЮЧЕНО → отключаем
	cleanupAll()
	print("Hitbox rainbovw: 🔴 Disabled")
else
	-- Состояние ВЫКЛЮЧЕНО → включаем
	local Storage = Instance.new("Folder")
	Storage.Name = "HighlightToggle"
	Storage.Parent = CoreGui

	local function getRainbowColor(timeOffset)
		local frequency = 1
		local r = math.sin(frequency * timeOffset + 0) * 127 + 128
		local g = math.sin(frequency * timeOffset + 2) * 127 + 128
		local b = math.sin(frequency * timeOffset + 4) * 127 + 128
		return Color3.fromRGB(math.floor(r), math.floor(g), math.floor(b))
	end

	local function createHighlight(plr)
		-- Себя не подсвечиваем
		if plr == lp then
			return
		end

		if playerHighlights[plr] then
			playerHighlights[plr]:Destroy()
			playerHighlights[plr] = nil
		end
		if playerConnections[plr] then
			playerConnections[plr]:Disconnect()
			playerConnections[plr] = nil
		end

		local Highlight = Instance.new("Highlight")
		Highlight.Name = plr.Name
		Highlight.DepthMode = DepthMode
		Highlight.FillTransparency = FillTransparency
		Highlight.OutlineColor = OutlineColor
		Highlight.OutlineTransparency = OutlineTransparency
		Highlight.Parent = Storage

		playerHighlights[plr] = Highlight

		if plr.Character then
			Highlight.Adornee = plr.Character
		end

		playerConnections[plr] = plr.CharacterAdded:Connect(function(char)
			if playerHighlights[plr] then
				playerHighlights[plr].Adornee = char
			end
		end)
	end

	-- Запускаем радугу
	rainbowConnection = RunService.Heartbeat:Connect(function()
		local time = tick()
		local currentColor = getRainbowColor(time)

		for plr, highlight in pairs(playerHighlights) do
			if plr ~= lp and highlight and highlight.Parent then
				highlight.FillColor = currentColor
			end
		end
	end)

	-- Подключаем всех игроков, кроме себя
	table.insert(globalConnections, Players.PlayerAdded:Connect(createHighlight))
	for _, v in ipairs(Players:GetPlayers()) do
		createHighlight(v)
	end

	-- Очистка при выходе игрока
	table.insert(globalConnections, Players.PlayerRemoving:Connect(function(plr)
		local highlight = playerHighlights[plr]
		if highlight then
			highlight:Destroy()
			playerHighlights[plr] = nil
		end

		if playerConnections[plr] then
			playerConnections[plr]:Disconnect()
			playerConnections[plr] = nil
		end
	end))

	print("Hitbox rainbow: 🟢 Enabled")
end
