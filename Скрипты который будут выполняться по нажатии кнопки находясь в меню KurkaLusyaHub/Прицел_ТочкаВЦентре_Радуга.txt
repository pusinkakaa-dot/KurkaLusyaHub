local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

-- toggle
local existing = playerGui:FindFirstChild("CenterDotGui")

if existing then
	existing:Destroy()
	print("Aiming dot in the center of the screen: 🔴 Disabled")
	return
end

print("Aiming dot in the center of the screen: 🟢 Enabled")

-- создаём GUI
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "CenterDotGui"
screenGui.IgnoreGuiInset = true
screenGui.Parent = playerGui

local dot = Instance.new("Frame")
dot.Name = "Dot"
dot.Parent = screenGui

-- центр
dot.Position = UDim2.new(0.5, 0, 0.5, 0)
dot.AnchorPoint = Vector2.new(0.5, 0.5)

-- 🔥 маленькая точка
dot.Size = UDim2.new(0, 3, 0, 3)

dot.BackgroundColor3 = Color3.new(1, 1, 1)
dot.BorderSizePixel = 0

-- круг
local corner = Instance.new("UICorner")
corner.CornerRadius = UDim.new(1, 0)
corner.Parent = dot

-- 🌈 радуга
local hue = 0

RunService.RenderStepped:Connect(function(dt)
	if not dot or not dot.Parent then return end

	hue += dt * 0.3
	if hue > 1 then hue = 0 end

	dot.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
end)
