local Players = game:GetService("Players")
local ContextActionService = game:GetService("ContextActionService")
local UserInputService = game:GetService("UserInputService")
local TweenService = game:GetService("TweenService")

local player = Players.LocalPlayer
local camera = workspace.CurrentCamera

if player:GetAttribute("FovToggle") == nil then
	player:SetAttribute("FovToggle", false)
end

local toggled = player:GetAttribute("FovToggle")
local newState = not toggled
player:SetAttribute("FovToggle", newState)
print(newState and "Camera zooming using the ALT key: 🟢 Enabled" or "Camera zooming using the ALT key: 🔴 Disabled")

local defaultFov = camera.FieldOfView
local zoomFov = 10
local tweenInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

local activeTween: Tween? = nil
local focusConn: RBXScriptConnection? = nil

local function tweenFov(targetFov: number)
	if activeTween then
		activeTween:Cancel()
		activeTween = nil
	end

	activeTween = TweenService:Create(camera, tweenInfo, { FieldOfView = targetFov })
	activeTween:Play()
end

local function resetFov()
	tweenFov(defaultFov)
end

if not newState then
	ContextActionService:UnbindAction("FovZoom")
	if focusConn then
		focusConn:Disconnect()
		focusConn = nil
	end
	camera.FieldOfView = defaultFov
	return
end

local function handle(actionName, inputState, inputObject)
	if inputState == Enum.UserInputState.Begin then
		tweenFov(zoomFov)
	elseif inputState == Enum.UserInputState.End then
		resetFov()
	end
	return Enum.ContextActionResult.Pass
end

focusConn = UserInputService.WindowFocusReleased:Connect(resetFov)

ContextActionService:BindAction(
	"FovZoom",
	handle,
	false,
	Enum.KeyCode.LeftAlt
)
