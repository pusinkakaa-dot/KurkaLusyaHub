local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")

local LocalPlayer = Players.LocalPlayer
if not LocalPlayer then return end

-- Настройки кольца на лице
local RING_GUI_NAME = "FaceRainbowRingGUI"
local RING_SIZE = UDim2.fromScale(1.30, 1.30) -- Удобный размер вокруг лица для попаданий
local RING_THICKNESS = 2                    -- Тонкое кольцо
local REFRESH_INTERVAL = 2.5                    -- Обновление каждые 5 секунд
local RAINBOW_SPEED = 0.6                     -- Скорость смены цвета радуги

-- Переключатель Вкл / Выкл при каждом запуске
local TOGGLE_ATTR = "FaceRainbowRingState"
local RUN_ID_ATTR = "FaceRainbowRingRunId"

local stateFolder = ReplicatedStorage:FindFirstChild("FaceRingStateFolder")
if not stateFolder then
	stateFolder = Instance.new("Folder")
	stateFolder.Name = "FaceRingStateFolder"
	stateFolder.Parent = ReplicatedStorage
end

local nextState = not (stateFolder:GetAttribute(TOGGLE_ATTR) == true)
local runId = (stateFolder:GetAttribute(RUN_ID_ATTR) or 0) + 1

stateFolder:SetAttribute(TOGGLE_ATTR, nextState)
stateFolder:SetAttribute(RUN_ID_ATTR, runId)

print("A rainbow ring placed over the faces of all players except you: " .. (nextState and "🟢 Enabled" or "🔴 Disabled"))

local function isActiveRun(): boolean
	return stateFolder:GetAttribute(TOGGLE_ATTR) == true
		and stateFolder:GetAttribute(RUN_ID_ATTR) == runId
end

local activeStrokes: { UIStroke } = {}
local renderConn: RBXScriptConnection? = nil

local function clearAllRings()
	table.clear(activeStrokes)
	for _, plr in ipairs(Players:GetPlayers()) do
		local char = plr.Character
		if char then
			for _, desc in ipairs(char:GetDescendants()) do
				if desc:IsA("BillboardGui") and desc.Name == RING_GUI_NAME then
					desc:Destroy()
				end
			end
		end
	end
end

local function stopAll()
	if renderConn then
		renderConn:Disconnect()
		renderConn = nil
	end
	clearAllRings()
end

if not nextState then
	stopAll()
	return
end

clearAllRings()

local function attachRingToHead(character: Model)
	local head = character:FindFirstChild("Head") :: BasePart?
	if not head then return end

	local hum = character:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return end

	local oldGui = head:FindFirstChild(RING_GUI_NAME)
	if oldGui then
		oldGui:Destroy()
	end

	-- BillboardGui крепится ровно к лицу/голове и всегда смотрит на камеру
	local billboard = Instance.new("BillboardGui")
	billboard.Name = RING_GUI_NAME
	billboard.Adornee = head
	billboard.AlwaysOnTop = true
	billboard.Size = RING_SIZE
	billboard.MaxDistance = 500
	billboard.Parent = head

	-- Невидимая кликабельная/целевая зона в центре кольца для удобного наведения и попаданий
	local hitArea = Instance.new("TextButton")
	hitArea.Name = "RingHitZone"
	hitArea.Size = UDim2.fromScale(1, 1)
	hitArea.BackgroundTransparency = 1
	hitArea.Text = ""
	hitArea.AutoButtonColor = false
	hitArea.Active = false -- Не блокирует стандартное управление и клики по миру/игроку
	hitArea.Parent = billboard

	-- Само тонкое кольцо через круглый Frame + UIStroke
	local ringFrame = Instance.new("Frame")
	ringFrame.Name = "RingVisual"
	ringFrame.AnchorPoint = Vector2.new(0.5, 0.5)
	ringFrame.Position = UDim2.fromScale(0.5, 0.5)
	ringFrame.Size = UDim2.fromScale(1, 1)
	ringFrame.BackgroundTransparency = 1
	ringFrame.Parent = hitArea

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(1, 0)
	corner.Parent = ringFrame

	local stroke = Instance.new("UIStroke")
	stroke.Name = "RainbowStroke"
	stroke.Thickness = RING_THICKNESS
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Color = Color3.fromHSV(0, 1, 1)
	stroke.Parent = ringFrame

	table.insert(activeStrokes, stroke)
end

local function refreshRings()
	clearAllRings()
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer and plr.Character then
			attachRingToHead(plr.Character)
		end
	end
end

-- Плавная радуга на каждом кадре
local hue = 0
renderConn = RunService.RenderStepped:Connect(function(dt: number)
	if not isActiveRun() then
		stopAll()
		return
	end

	hue = (hue + dt * RAINBOW_SPEED) % 1
	local rainbowColor = Color3.fromHSV(hue, 1, 1)

	for _, stroke in ipairs(activeStrokes) do
		if stroke and stroke.Parent then
			stroke.Color = rainbowColor
		end
	end
end)

-- Обновление списка игроков и лиц каждые 5 секунд
task.spawn(function()
	while isActiveRun() do
		refreshRings()

		local elapsed = 0
		while elapsed < REFRESH_INTERVAL do
			if not isActiveRun() then
				stopAll()
				return
			end
			elapsed += task.wait(0.1)
		end
	end

	stopAll()
end)
