local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")

local LocalPlayer = Players.LocalPlayer
if not LocalPlayer then return end

-- Настройки квадратного блока-камеры, клавиши ("э" / Quote), метки и полёта
local TOGGLE_KEY = Enum.KeyCode.Quote
local PART_SIZE = Vector3.new(2, 2, 2)
local HEAD_OFFSET = Vector3.new(0, 4.5, 0)
local MARKER_TEXT = "⬇️"
local MARKER_OFFSET = Vector3.new(0, 4.2, 0)
local FLY_SPEED = 100
local SPRINT_MULTIPLIER = 2.4
local VERTICAL_SPEED = 28
local MOUSE_SENSITIVITY = 0.0035

-- Переключатель сессии самого скрипта при повторном запуске кода
local SESSION_ATTR = "HeadDroneCamQuoteSessionActive"
local SESSION_ID_ATTR = "HeadDroneCamQuoteSessionId"

local stateFolder = ReplicatedStorage:FindFirstChild("HeadDroneCamQuoteFolder")
if not stateFolder then
	stateFolder = Instance.new("Folder")
	stateFolder.Name = "HeadDroneCamQuoteFolder"
	stateFolder.Parent = ReplicatedStorage
end

local nextSessionActive = not (stateFolder:GetAttribute(SESSION_ATTR) == true)
local sessionId = (stateFolder:GetAttribute(SESSION_ID_ATTR) or 0) + 1

stateFolder:SetAttribute(SESSION_ATTR, nextSessionActive)
stateFolder:SetAttribute(SESSION_ID_ATTR, sessionId)

print("FreeCameraFree camera (activated by the ' key): " .. (nextSessionActive and "🟢 Enabled" or "🔴 Disabled"))

local function isSessionActive(): boolean
	return stateFolder:GetAttribute(SESSION_ATTR) == true
		and stateFolder:GetAttribute(SESSION_ID_ATTR) == sessionId
end

local listenerConnections: { RBXScriptConnection } = {}
local activeModeConnections: { RBXScriptConnection } = {}

local isDroneEnabled = false
local dronePart: Part? = nil
local playerPinGui: BillboardGui? = nil
local savedCameraType: Enum.CameraType? = nil
local savedCameraSubject: Instance? = nil
local savedWalkSpeed: number? = nil
local savedJumpPower: number? = nil
local savedUseJumpPower: boolean? = nil
local savedAutoRotate: boolean? = nil
local savedRootAnchored: boolean? = nil
local savedMouseBehavior: Enum.MouseBehavior? = nil
local savedMouseIconEnabled: boolean? = nil

local function setCenterMouseLock(enabled: boolean)
	pcall(function()
		if enabled then
			UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
			UserInputService.MouseIconEnabled = false
		else
			UserInputService.MouseBehavior = savedMouseBehavior or Enum.MouseBehavior.Default
			UserInputService.MouseIconEnabled = if savedMouseIconEnabled ~= nil then savedMouseIconEnabled else true
		end
	end)
end

local function clearModeConnections()
	for _, conn in ipairs(activeModeConnections) do
		conn:Disconnect()
	end
	table.clear(activeModeConnections)
end

local function removePlayerPin()
	if playerPinGui then
		pcall(function()
			playerPinGui:Destroy()
		end)
		playerPinGui = nil
	end

	local char = LocalPlayer.Character
	local head = char and char:FindFirstChild("Head")
	local root = char and char:FindFirstChild("HumanoidRootPart")
	for _, parentObj in ipairs({ head, root }) do
		if parentObj then
			local existing = parentObj:FindFirstChild("DroneBodyPinMarker")
			if existing then
				pcall(function()
					existing:Destroy()
				end)
			end
		end
	end
end

local function createPlayerPin(adornee: BasePart)
	removePlayerPin()

	local billboard = Instance.new("BillboardGui")
	billboard.Name = "DroneBodyPinMarker"
	billboard.Adornee = adornee
	billboard.Size = UDim2.fromOffset(54, 54)
	billboard.StudsOffset = MARKER_OFFSET
	billboard.AlwaysOnTop = true
	billboard.MaxDistance = math.huge
	billboard.Parent = adornee

	local label = Instance.new("TextLabel")
	label.Name = "PinLabel"
	label.Size = UDim2.fromScale(1, 1)
	label.BackgroundTransparency = 1
	label.Text = MARKER_TEXT
	label.TextScaled = true
	label.Font = Enum.Font.GothamBold
	label.Parent = billboard

	playerPinGui = billboard
end

local function disableDroneMode()
	isDroneEnabled = false
	clearModeConnections()
	removePlayerPin()

	local camera = workspace.CurrentCamera
	local char = LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local root = char and char:FindFirstChild("HumanoidRootPart") :: BasePart?

	if hum then
		if savedWalkSpeed ~= nil then hum.WalkSpeed = savedWalkSpeed end
		if savedJumpPower ~= nil then hum.JumpPower = savedJumpPower end
		if savedUseJumpPower ~= nil then hum.UseJumpPower = savedUseJumpPower end
		if savedAutoRotate ~= nil then hum.AutoRotate = savedAutoRotate end
	end

	if root and savedRootAnchored ~= nil then
		root.Anchored = savedRootAnchored
	end

	if camera then
		camera.CameraType = savedCameraType or Enum.CameraType.Custom
		if savedCameraSubject and savedCameraSubject.Parent then
			camera.CameraSubject = savedCameraSubject
		elseif hum then
			camera.CameraSubject = hum
		end
	end

	setCenterMouseLock(false)

	if dronePart then
		dronePart:Destroy()
		dronePart = nil
	end

	for _, child in ipairs(workspace:GetChildren()) do
		if child.Name == "HeadDroneCamPart" then
			child:Destroy()
		end
	end

	savedCameraType = nil
	savedCameraSubject = nil
	savedWalkSpeed = nil
	savedJumpPower = nil
	savedUseJumpPower = nil
	savedAutoRotate = nil
	savedRootAnchored = nil
	savedMouseBehavior = nil
	savedMouseIconEnabled = nil
end

local function enableDroneMode()
	if not isSessionActive() or isDroneEnabled then return end

	local char = LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local root = char and char:FindFirstChild("HumanoidRootPart") :: BasePart?
	local head = char and char:FindFirstChild("Head") :: BasePart?
	local camera = workspace.CurrentCamera

	if not char or not hum or not root or not camera then
		warn("Free camera: character or camera not found")
		return
	end

	isDroneEnabled = true

	savedCameraType = camera.CameraType
	savedCameraSubject = camera.CameraSubject
	savedWalkSpeed = hum.WalkSpeed
	savedJumpPower = hum.JumpPower
	savedUseJumpPower = hum.UseJumpPower
	savedAutoRotate = hum.AutoRotate
	savedRootAnchored = root.Anchored
	savedMouseBehavior = UserInputService.MouseBehavior
	savedMouseIconEnabled = UserInputService.MouseIconEnabled

	hum.WalkSpeed = 0
	hum.JumpPower = 0
	hum.AutoRotate = false
	root.AssemblyLinearVelocity = Vector3.zero
	root.AssemblyAngularVelocity = Vector3.zero
	root.Anchored = true

	local pinTarget = head or root
	createPlayerPin(pinTarget)

	local startPos = pinTarget.Position + HEAD_OFFSET
	local part = Instance.new("Part")
	part.Name = "HeadDroneCamPart"
	part.Shape = Enum.PartType.Block
	part.Size = PART_SIZE
	part.Position = startPos
	part.Anchored = true
	part.CanCollide = true
	part.CanTouch = false
	part.CanQuery = false
	part.Material = Enum.Material.Neon
	part.Color = Color3.fromRGB(0, 210, 255)
	part.Transparency = 0.25
	part.Parent = workspace
	dronePart = part

	camera.CameraType = Enum.CameraType.Scriptable
	camera.CameraSubject = part
	setCenterMouseLock(true)

	local yaw = 0
	local pitch = 0
	do
		local look = camera.CFrame.LookVector
		yaw = math.atan2(-look.X, -look.Z)
		pitch = math.asin(math.clamp(look.Y, -0.99, 0.99))
	end

	table.insert(activeModeConnections, UserInputService.InputChanged:Connect(function(input: InputObject)
		if not isSessionActive() or not isDroneEnabled then return end
		if input.UserInputType == Enum.UserInputType.MouseMovement then
			yaw -= input.Delta.X * MOUSE_SENSITIVITY
			pitch = math.clamp(pitch - input.Delta.Y * MOUSE_SENSITIVITY, math.rad(-85), math.rad(85))
		end
	end))

	table.insert(activeModeConnections, RunService.RenderStepped:Connect(function(dt: number)
		if not isSessionActive() then
			disableDroneMode()
			return
		end
		if not isDroneEnabled then return end

		local cam = workspace.CurrentCamera
		if not cam or not dronePart or not dronePart.Parent then return end

		if UserInputService.MouseBehavior ~= Enum.MouseBehavior.LockCenter then
			setCenterMouseLock(true)
		end

		local rotCFrame = CFrame.fromOrientation(pitch, yaw, 0)
		local forward = rotCFrame.LookVector
		local right = rotCFrame.RightVector

		local moveDir = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then
			moveDir += forward
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then
			moveDir -= forward
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then
			moveDir += right
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then
			moveDir -= right
		end

		if moveDir.Magnitude > 0 then
			moveDir = moveDir.Unit
		end

		local vertical = 0
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) or UserInputService:IsKeyDown(Enum.KeyCode.E) then
			vertical += 1
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.Q) then
			vertical -= 1
		end

		local speedMult = UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) and SPRINT_MULTIPLIER or 1
		local deltaPos = (moveDir * FLY_SPEED * speedMult + Vector3.new(0, vertical * VERTICAL_SPEED * speedMult, 0)) * dt

		local nextPos = dronePart.Position + deltaPos
		dronePart.CFrame = CFrame.new(nextPos) * rotCFrame
		cam.CameraType = Enum.CameraType.Scriptable
		cam.CFrame = dronePart.CFrame
	end))
end

local function toggleDroneMode()
	if not isSessionActive() then return end
	if isDroneEnabled then
		disableDroneMode()
		print("FreeCamera: 🔴 Disabled")
	else
		enableDroneMode()
		print("FreeCamera: 🟢 Enabled")
	end
end

local function stopSession()
	disableDroneMode()
	for _, conn in ipairs(listenerConnections) do
		conn:Disconnect()
	end
	table.clear(listenerConnections)
end

if not nextSessionActive then
	stopSession()
	return
end

stopSession()

table.insert(listenerConnections, UserInputService.InputBegan:Connect(function(input: InputObject, gameProcessed: boolean)
	if not isSessionActive() then
		stopSession()
		return
	end
	if gameProcessed then return end
	if input.KeyCode == TOGGLE_KEY then
		toggleDroneMode()
	end
end))

table.insert(listenerConnections, LocalPlayer.CharacterAdded:Connect(function()
	if isDroneEnabled then
		disableDroneMode()
	end
end))
